% KKSS application icon: a tessellated kiss, inspired by the teal faces and
% white mesh edges of Kratos Multiphysics. Keep the polygon mesh vector-native
% so every installer/window size is rendered from the same editable source.
\documentclass[tikz,border=0mm]{standalone}
\definecolor{kkssbg}{HTML}{14191F}
\definecolor{kratosteal}{HTML}{008688}
\definecolor{teallight}{HTML}{11999A}
\definecolor{tealdark}{HTML}{007477}
\begin{document}
\begin{tikzpicture}[x=1mm,y=1mm,line cap=round,line join=round]
  \fill[kkssbg,rounded corners=5.5mm] (-14,-14) rectangle (14,14);

  % Shared nodes form a conforming triangular mesh: a pronounced cupid's bow
  % above, a full lower lip below, and a narrow open kiss between them.
  \coordinate (L) at (-11.5,0);
  \coordinate (A) at (-7.5,4.2);
  \coordinate (B) at (-3.8,6.7);
  \coordinate (C) at (0,4.8);
  \coordinate (D) at (3.8,6.7);
  \coordinate (E) at (7.5,4.2);
  \coordinate (R) at (11.5,0);
  \coordinate (U) at (-5.3,1.2);
  \coordinate (V) at (0,0.5);
  \coordinate (W) at (5.3,1.2);
  \coordinate (I) at (-5.3,-1.2);
  \coordinate (J) at (0,-1.8);
  \coordinate (K) at (5.3,-1.2);
  \coordinate (F) at (-7.4,-4.5);
  \coordinate (G) at (-3.7,-6.5);
  \coordinate (H) at (0,-7.1);
  \coordinate (M) at (3.7,-6.5);
  \coordinate (N) at (7.4,-4.5);

  % Fill facets first; draw each shared edge just once for even mesh weights.
  \fill[kratosteal] (L)--(A)--(B)--(C)--(D)--(E)--(R)--(W)--(V)--(U)--cycle;
  \fill[teallight] (A)--(B)--(U)--cycle (B)--(C)--(V)--cycle (C)--(D)--(W)--cycle;
  \fill[tealdark] (L)--(A)--(U)--cycle (U)--(B)--(V)--cycle (W)--(E)--(R)--cycle;
  \fill[kratosteal] (L)--(I)--(J)--(K)--(R)--(N)--(M)--(H)--(G)--(F)--cycle;
  \fill[teallight] (L)--(I)--(F)--cycle (I)--(J)--(H)--cycle (J)--(K)--(M)--cycle;
  \fill[tealdark] (F)--(G)--(I)--cycle (G)--(H)--(I)--cycle (K)--(N)--(M)--cycle;

  \draw[white,line width=0.20mm]
    (A)--(U)--(B)--(V)--(C)--(W)--(D)
    (E)--(W)
    (F)--(I)--(G) (I)--(H)--(J)--(M)--(K)--(N);
  % A stronger white silhouette echoes the Kratos reference and preserves the
  % kiss shape at small sizes. The mouth opening remains the backdrop color.
  \draw[white,line width=0.48mm]
    (L)--(A)--(B)--(C)--(D)--(E)--(R)--(W)--(V)--(U)--cycle
    (L)--(I)--(J)--(K)--(R)--(N)--(M)--(H)--(G)--(F)--cycle;
\end{tikzpicture}
\end{document}
